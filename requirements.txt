# PediAid Voice Studio — pinned to the environment proven on macOS Intel x86_64.
# Do not bump these casually: torch 2.2.2 / numpy 1.26.4 / transformers 4.49.0
# is a known-good combination for kokoro 0.9.4 on this machine.

# --- desktop UI ---
PySide6==6.11.1

# --- local speech synthesis ---
kokoro==0.9.4
misaki[en]==0.9.4
torch==2.2.2
# kokoro 0.9.4 targets the transformers 4.x API; 5.x changes it.
transformers==4.49.0

# --- local speech recognition ---
# Writes the script from a video that already has a voice on it. Chosen over
# openai-whisper because it runs on CTranslate2 and needs no PyTorch of its own,
# so it changes none of the pins above.
#
# It does bring a second Intel OpenMP runtime alongside torch's, and two of
# those in one process abort it when a model loads — which is why transcription
# runs in its own process. See app/transcribe/worker_main.py.
faster-whisper==1.1.1
ctranslate2==4.8.1

# --- audio ---
numpy==1.26.4
soundfile==0.14.0

# PyAV is FFmpeg, compiled into the wheel. Everything that used to shell out to
# an ffmpeg binary — reading audio out of a video, fitting speech to its window,
# writing the MP3 — runs against these libraries instead, so a new user is never
# sent to Homebrew before the app will work. faster-whisper depends on it too;
# it is pinned here because the app now relies on it directly.
av==18.1.0

# --- tests ---
pytest==8.3.4
